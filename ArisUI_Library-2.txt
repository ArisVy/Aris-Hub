local function GetRGB()
    return Color3.fromHSV(tick() % 5 / 5, 1, 1)
end

local function GetMovingColorSequence(palette, shift)
    local keypoints = {}
    for step = 0, 5 do
        local i = step / 5
        local t = (i - shift) % 1
        if t < 0 then t = t + 1 end
        local segments = #palette - 1
        local scaled = t * segments
        local index = math.floor(scaled) + 1
        local fraction = scaled - (index - 1)
        local color
        if index >= #palette then
            color = palette[#palette]
        else
            color = palette[index]:Lerp(palette[index + 1], fraction)
        end
        table.insert(keypoints, ColorSequenceKeypoint.new(i, color))
    end
    return ColorSequence.new(keypoints)
end

local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            pcall(function() inst[k] = v end)
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function Tween(obj, props, duration, style, dir)
    duration = (duration or 0.25) / (Library.Settings.AnimationSpeed or 1)
    style = style or Enum.EasingStyle.Quad
    dir = dir or Enum.EasingDirection.Out
    local tw = TweenService:Create(obj, TweenInfo.new(duration, style, dir), props)
    tw:Play()
    return tw
end

local function MakeDraggable(frame, dragHandle)
    dragHandle = dragHandle or frame
    local dragging, dragStart, startPos

    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    dragHandle.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function ApplyButtonAnimation(btn)
    local scale = btn:FindFirstChildOfClass("UIScale") or Create("UIScale", {Parent = btn, Scale = 1})
    local hoverScale = Library.Settings.ButtonHoverScale or 1.05
    local clickScale = Library.Settings.ButtonClickScale or 0.92

    btn.MouseEnter:Connect(function()
        Tween(scale, {Scale = hoverScale}, 0.15)
    end)
    btn.MouseLeave:Connect(function()
        Tween(scale, {Scale = 1}, 0.15)
    end)
    btn.MouseButton1Down:Connect(function()
        Tween(scale, {Scale = clickScale}, 0.08)
    end)
    btn.MouseButton1Up:Connect(function()
        Tween(scale, {Scale = hoverScale}, 0.1, Enum.EasingStyle.Back)
    end)
end

        Rotation = 90
    })
    table.insert(Library._AllStrokes, stroke)
    table.insert(Library._AllGradients, grad)
    return stroke, grad
end

local function ApplyToggleGradient(parent, isOn)
    local grad = parent:FindFirstChild("ToggleGrad")
    if not grad then
        grad = Create("UIGradient", {
            Parent = parent,
            Name = "ToggleGrad",
            Rotation = 90
        })
        table.insert(Library._AllGradients, grad)
    end
    parent:SetAttribute("IsOn", isOn and true or false)

    local txt = parent:FindFirstChildOfClass("TextLabel")
    if txt then
        local txtGrad = txt:FindFirstChildOfClass("UIGradient")
        if isOn then
            if txtGrad then
                txtGrad.Enabled = true
                txtGrad.Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 0, 0)),
                    ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
                })
                txtGrad:SetAttribute("CustomOnColor", true)
            end
            txt.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            if txtGrad then
                txtGrad.Enabled = true
                txtGrad:SetAttribute("CustomOnColor", false)
            end
            txt.TextColor3 = Color3.fromRGB(255, 255, 255)
        end
    end
end

local function EnsureConfigFolder()
    if not isfolder then return false end
    if not isfolder(Library.ConfigFolder) then
        pcall(makefolder, Library.ConfigFolder)
    end
    return true
end

function Library:SaveConfig(name)
    name = name or Library.CurrentConfig or "Default"
    if not EnsureConfigFolder() then
        Library:Notify({Title = "Config", Description = "Executor không hỗ trợ file system", Duration = 3})
        return false
    end

    local data = {
        Flags = {},
        Settings = Library.Settings,
        Theme = Library.CurrentThemeName
    }
    for flag, value in pairs(Library.Flags) do
        data.Flags[flag] = value
    end

    local success, err = pcall(function()
        writefile(Library.ConfigFolder .. "/" .. name .. ".json", HttpService:JSONEncode(data))
    end)

    if success then
        Library.CurrentConfig = name
        Library:Notify({Title = "Config", Description = "Đã lưu: " .. name, Duration = 2})
        return true
    else
        Library:Notify({Title = "Config", Description = "Lỗi lưu: " .. tostring(err), Duration = 3})
        return false
    end
end

function Library:LoadConfig(name)
    name = name or Library.CurrentConfig or "Default"
    if not EnsureConfigFolder() then return false end

    local path = Library.ConfigFolder .. "/" .. name .. ".json"
    if not isfile or not isfile(path) then
        Library:Notify({Title = "Config", Description = "Không tìm thấy: " .. name, Duration = 3})
        return false
    end

    local success, data = pcall(function()
        return HttpService:JSONDecode(readfile(path))
    end)

    if not success or type(data) ~= "table" then
        Library:Notify({Title = "Config", Description = "Config bị lỗi", Duration = 3})
        return false
    end

    -- Load theme
    if data.Theme and Library.Themes[data.Theme] then
        Library:SetTheme(data.Theme)
    end

    -- Load settings
    if data.Settings then
        for k, v in pairs(data.Settings) do
            Library.Settings[k] = v
        end
    end

    -- Load flags
    if data.Flags then
        for flag, value in pairs(data.Flags) do
            Library.Flags[flag] = value
            if Library._FlagElements[flag] and Library._FlagElements[flag].Set then
                pcall(Library._FlagElements[flag].Set, value)
            end
        end
    end

    Library.CurrentConfig = name
    Library:Notify({Title = "Config", Description = "Đã load: " .. name, Duration = 2})
    return true
end

function Library:DeleteConfig(name)
    if not name or not EnsureConfigFolder() then return false end
    local path = Library.ConfigFolder .. "/" .. name .. ".json"
    if isfile and isfile(path) then
        pcall(delfile, path)
        Library:Notify({Title = "Config", Description = "Đã xóa: " .. name, Duration = 2})
        return true
    end
    return false
end

function Library:GetConfigList()
    if not EnsureConfigFolder() or not listfiles then return {} end
    local list = {}
    for _, file in ipairs(listfiles(Library.ConfigFolder)) do
        local n = file:match("([^/\\]+)%.json$")
        if n then table.insert(list, n) end
    end
    return list
end

function Library:SetFlag(flag, value)
    Library.Flags[flag] = value
    if Library.AutoSave then
        task.spawn(function()
            Library:SaveConfig(Library.CurrentConfig)
        end)
    end
end

function Library:SetTheme(themeName)
    if not Library.Themes[themeName] then return false end
    Library.Theme = Library.Themes[themeName]
    Library.CurrentThemeName = themeName

    -- Update existing windows
    for _, win in ipairs(Library.Windows) do
        if win.MainFrame then
            win.MainFrame.BackgroundColor3 = Library.Theme.Background
            AddAcrylic(win.MainFrame)
        end
    end

    Library:Notify({Title = "Theme", Description = "Đã đổi sang: " .. Library.Theme.Name, Duration = 2})
    return true
end

function Library:CreateCustomTheme(name, themeTable)
    Library.Themes[name] = themeTable
    themeTable.Name = name
end

            local flag = opts.Flag
            local callback = opts.Callback or function() end

            if flag then
                if Library.Flags[flag] ~= nil then

                else

        function Tab:CreateSlider(opts)
            opts = opts or {}
            local name = opts.Name or "Slider"
            local min = opts.Min or 0
            local max = opts.Max or 100
            local default = opts.Default or min
            local flag = opts.Flag
            local callback = opts.Callback or function() end
            local step = opts.Step or 1

            if flag then
                if Library.Flags[flag] ~= nil then default = Library.Flags[flag]
                else Library.Flags[flag] = default end
            end

            local container = Create("Frame", {
                Parent = content,
                Size = UDim2.new(1, -8, 0, 72),
                BackgroundTransparency = 1,
                LayoutOrder = #Tab.Elements + 1
            })

            local title = Create("TextLabel", {
                Parent = container,
                Size = UDim2.new(1, 0, 0, 22),
                BackgroundTransparency = 1,
                Text = name .. ": " .. tostring(default),
                Font = Enum.Font.GothamBold,
                TextSize = 13,
                TextColor3 = Color3.new(1, 1, 1),
                TextXAlignment = Enum.TextXAlignment.Left
            })
            CreateTextGradient(title)

            local sliderBg = Create("Frame", {
                Parent = container,
                Size = UDim2.new(1, 0, 0, 26),
                Position = UDim2.new(0, 0, 0, 30),
                BackgroundColor3 = Color3.fromRGB(22, 22, 26)
            })
            Create("UICorner", {Parent = sliderBg, CornerRadius = UDim.new(0, 14)})
            CreateBorder(sliderBg, 1.2)

            local fill = Create("Frame", {
                Parent = sliderBg,
                Size = UDim2.new(math.clamp((default - min) / (max - min), 0, 1), 0, 1, 0),
                BackgroundColor3 = Color3.new(1, 1, 1)
            })
            Create("UICorner", {Parent = fill, CornerRadius = UDim.new(0, 14)})
            ApplyToggleGradient(fill, true)

            local value = default

            local function Set(val)
                val = math.clamp(tonumber(val) or min, min, max)
                if step then val = math.floor(val / step + 0.5) * step end
                value = val
                local ratio = (value - min) / math.max(max - min, 0.001)
                fill.Size = UDim2.new(ratio, 0, 1, 0)
                title.Text = name .. ": " .. tostring(value)
                if flag then Library:SetFlag(flag, value) end
                task.spawn(callback, value)
            end

            local dragging = false
            sliderBg.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                end
            end)
            UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = false
                end
            end)
            UserInputService.InputChanged:Connect(function(input)
                if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    local rel = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
                    Set(min + rel * (max - min))
                end
            end)

            if flag then
                Library._FlagElements[flag] = {Set = Set, Get = function() return value end}
            end

            table.insert(Tab.Elements, container)
            return {Set = Set, Get = function() return value end}
        end

            local selected = multi and (type(default) == "table" and default or {}) or default
            local isOpen = false

            local function Refresh()
                if multi then
                    btnTxt.Text = name .. ": " .. (#selected > 0 and table.concat(selected, ", ") or "None")
                else
                    btnTxt.Text = name .. ": " .. tostring(selected)
                end
            end

            local function ToggleDrop()
                isOpen = not isOpen
                dropFrame.Visible = isOpen
                if isOpen then
                    local h = math.min(#options * 34 + 14, 180)
                    Tween(dropFrame, {Size = UDim2.new(1, 0, 0, h)}, 0.22)
                    container.Size = UDim2.new(1, -8, 0, 38 + h + 10)
                else
                    Tween(dropFrame, {Size = UDim2.new(1, 0, 0, 0)}, 0.18)
                    task.delay(0.2, function()
                        if not isOpen then container.Size = UDim2.new(1, -8, 0, 38) end
                    end)
                end
            end

            mainBtn.MouseButton1Click:Connect(ToggleDrop)

            for _, opt in ipairs(options) do
                local optBtn = Create("TextButton", {
                    Parent = dropFrame,
                    Size = UDim2.new(1, 0, 0, 30),
                    BackgroundColor3 = Color3.fromRGB(35, 35, 42),

                    TextColor3 = Color3.new(1, 1, 1),
                    ZIndex = 62

                        if idx then table.remove(selected, idx) else table.insert(selected, opt) end
                        if flag then Library:SetFlag(flag, selected) end
                        task.spawn(callback, selected)
                    else
                        selected = opt
                        if flag then Library:SetFlag(flag, selected) end
                        task.spawn(callback, selected)
                        ToggleDrop()
                    end
                    Refresh()
                end)
            end

            if flag then
                Library._FlagElements[flag] = {
                    Set = function(v) selected = v Refresh() end,
                    Get = function() return selected end
                }
            end

            table.insert(Tab.Elements, container)
            return {Set = function(v) selected = v Refresh() end, Get = function() return selected end}
        end

                    Library.Settings.GlowEnabled = v
                end
            })

            self:CreateToggle({
                Name = "Acrylic Effect",
                Default = Library.Settings.AcrylicEnabled,
                Flag = "UI_Acrylic",
                Callback = function(v)
                    Library.Settings.AcrylicEnabled = v
                end
            })

            self:CreateSlider({
                Name = "Gradient Speed",
                Min = 0.05,
                Max = 0.4,
                Default = Library.Settings.GradientSpeed,
                Step = 0.01,
                Flag = "UI_GradSpeed",
                Callback = function(v)
                    Library.Settings.GradientSpeed = v
                end
            })
        end

        return Tab
    end

    -- Gradient animation loop
    task.spawn(function()
        while ScreenGui and ScreenGui.Parent do
            local shift = tick() * (Library.Settings.GradientSpeed or 0.15)
            local seqOn = GetMovingColorSequence(Library.Theme.Accent, shift)
            local seqOff = GetMovingColorSequence(Library.Theme.Off, shift)
            local seqText = GetMovingColorSequence(Library.Theme.TextGradient, shift * 1.4)
            local seqBorder = GetMovingColorSequence(Library.Theme.Border, shift * 1.7)

            for _, obj in ipairs(MainFrame:GetDescendants()) do
                if obj:IsA("UIGradient") then
                    local parent = obj.Parent
                    if parent and parent:GetAttribute("IsOn") == true then
                        obj.Color = seqOn
                    elseif parent and parent:IsA("UIStroke") then
                        obj.Color = seqBorder
                    elseif not obj:GetAttribute("CustomOnColor") then
                        obj.Color = seqText
                    end
                end
            end
            task.wait(0.03)
        end
    end)

    if Library.AutoLoad then
        task.spawn(function()
            task.wait(0.6)
            Library:LoadConfig(Library.CurrentConfig)
        end)
    end

    table.insert(Library.Windows, Window)
    return Window
end

local function CloneTheme(base, overrides)
    local t = {}
    for k, v in pairs(base) do
        if type(v) == "table" then
            t[k] = {}
            for i, c in ipairs(v) do t[k][i] = c end
        else
            t[k] = v
        end
    end
    for k, v in pairs(overrides or {}) do
        t[k] = v
    end
    return t
end

    end
